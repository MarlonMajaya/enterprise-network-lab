! R1 - upstream/edge router
! Connects to R2 via a point-to-point /30 link, part of OSPF area 0

hostname R1
ip domain-name marlonlab.local

enable secret <REDACTED>

username admin privilege 15 secret <REDACTED>

! Interface to R2
interface g0/0
 description Link-to-R2
 ip address 10.0.0.1 255.255.255.252
 no shutdown

! OSPF
router ospf 1
 router-id 1.1.1.1
 network 10.0.0.0 0.0.0.3 area 0
 passive-interface default
 no passive-interface g0/0

! SSH access
crypto key generate rsa modulus 2048
ip ssh version 2

line vty 0 4
 login local
 transport input ssh

line console 0
 password <REDACTED>
 login

service password-encryption
